cmake_minimum_required(VERSION 3.16)
project(anakin_mc VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(anakin_sdk 0.1.0 EXACT REQUIRED)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/anakin_sdk.revision" SDK_LOCK LIMIT_COUNT 1)
if(NOT SDK_LOCK STREQUAL ANAKIN_SDK_REVISION)
  message(FATAL_ERROR "Installed anakin_sdk revision does not match anakin_sdk.revision")
endif()
set(ONNXRUNTIME_ROOT "" CACHE PATH "Official ONNX Runtime C++ 1.20.1 distribution")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(ORT_LIBRARY onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
file(READ "${ONNXRUNTIME_ROOT}/VERSION_NUMBER" ORT_VERSION)
string(STRIP "${ORT_VERSION}" ORT_VERSION)
if(NOT ORT_VERSION STREQUAL "1.20.1")
  message(FATAL_ERROR "Expected official ONNX Runtime C++ 1.20.1")
endif()
add_library(anakin_policy STATIC src/policy.cpp)
target_include_directories(anakin_policy PUBLIC include "${ORT_INCLUDE}")
ament_target_dependencies(anakin_policy anakin_sdk)
target_link_libraries(anakin_policy "${ORT_LIBRARY}")
set_target_properties(anakin_policy PROPERTIES BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib" INSTALL_RPATH "${ONNXRUNTIME_ROOT}/lib")
add_executable(anakin_mc_node src/node.cpp)
target_link_libraries(anakin_mc_node anakin_policy)
ament_target_dependencies(anakin_mc_node rclcpp anakin_sdk)
set_target_properties(anakin_mc_node PROPERTIES INSTALL_RPATH "${ONNXRUNTIME_ROOT}/lib")
add_executable(anakin_mc_trace tests/trace.cpp)
target_link_libraries(anakin_mc_trace anakin_policy)
ament_target_dependencies(anakin_mc_trace anakin_sdk)
set_target_properties(anakin_mc_trace PROPERTIES INSTALL_RPATH "${ONNXRUNTIME_ROOT}/lib")
add_executable(anakin_mc_transition_trace tests/transition_trace.cpp)
target_link_libraries(anakin_mc_transition_trace anakin_policy)
ament_target_dependencies(anakin_mc_transition_trace anakin_sdk)
set_target_properties(anakin_mc_transition_trace PROPERTIES INSTALL_RPATH "${ONNXRUNTIME_ROOT}/lib")
add_executable(anakin_mc_startup_test tests/startup.cpp)
target_link_libraries(anakin_mc_startup_test anakin_policy)
ament_target_dependencies(anakin_mc_startup_test anakin_sdk)
if(BUILD_TESTING)
 enable_testing()
 add_test(NAME startup_mux_contracts COMMAND anakin_mc_startup_test)
endif()
install(TARGETS anakin_mc_node anakin_mc_trace anakin_mc_transition_trace DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY docs/ DESTINATION share/${PROJECT_NAME}/docs)
install(FILES LICENSE LICENSE-Anakin NOTICE.md DESTINATION share/${PROJECT_NAME})
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/runtime-sdk.revision" "${SDK_LOCK}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/runtime-sdk.revision" DESTINATION share/${PROJECT_NAME})
ament_package()
